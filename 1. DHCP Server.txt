1. DHCP Server 

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <arpa/inet.h>

int main(int argc, char *argv[])
{
    int sockfd, connfd;
    struct sockaddr_in server, client;
    socklen_t len;
    char request[100];
    char ip[100];

    if (argc != 2) {
        printf("Usage: %s <port>\n", argv[0]);
        return 1;
    }

    sockfd = socket(AF_INET, SOCK_STREAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(atoi(argv[1]));
    server.sin_addr.s_addr = INADDR_ANY;

    bind(sockfd, (struct sockaddr *)&server, sizeof(server));
    listen(sockfd, 5);

    printf("DHCP Server waiting...\n");

    len = sizeof(client);
    connfd = accept(sockfd, (struct sockaddr *)&client, &len);

    recv(connfd, request, sizeof(request) - 1, 0);
    request[sizeof(request) - 1] = '\0';

    printf("Received request: %s\n", request);

    if (strcmp(request, "DHCPDISCOVER") == 0) {
        strcpy(ip, "192.168.1.100");

        send(connfd, ip, strlen(ip) + 1, 0);

        printf("IP Address assigned: %s\n", ip);
    }
    else {
        strcpy(ip, "Invalid Request");
        send(connfd, ip, strlen(ip) + 1, 0);
    }

    close(connfd);
    close(sockfd);

    return 0;
}


 DHCP client

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <arpa/inet.h>

int main(int argc, char *argv[])
{
    int sockfd;
    struct sockaddr_in server;
    char request[] = "DHCPDISCOVER";
    char ip[100];

    if (argc != 3) {
        printf("Usage: %s <IP> <port>\n", argv[0]);
        return 1;
    }

    sockfd = socket(AF_INET, SOCK_STREAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(atoi(argv[2]));
    inet_pton(AF_INET, argv[1], &server.sin_addr);

    connect(sockfd, (struct sockaddr *)&server, sizeof(server));

    printf("Sending DHCP Discover Request...\n");

    send(sockfd, request, strlen(request) + 1, 0);

    recv(sockfd, ip, sizeof(ip) - 1, 0);
    ip[sizeof(ip) - 1] = '\0';

    printf("Received IP Address: %s\n", ip);

    close(sockfd);

    return 0;
}
