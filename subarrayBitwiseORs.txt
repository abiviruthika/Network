int subarrayBitwiseORs(int* arr, int arrSize)
{
    int result[100000];
    int prev[32], curr[32];

    int resultCount = 0;
    int prevCount = 0;

    for (int i = 0; i < arrSize; i++)
    {
        int currCount = 0;

        curr[currCount++] = arr[i];

        for (int j = 0; j < prevCount; j++)
        {
            int val = prev[j] | arr[i];
            int found = 0;

            for (int k = 0; k < currCount; k++)
            {
                if (curr[k] == val)
                {
                    found = 1;
                    break;
                }
            }

            if (!found)
                curr[currCount++] = val;
        }

        for (int j = 0; j < currCount; j++)
        {
            int found = 0;

            for (int k = 0; k < resultCount; k++)
            {
                if (result[k] == curr[j])
                {
                    found = 1;
                    break;
                }
            }

            if (!found)
                result[resultCount++] = curr[j];
        }

        prevCount = currCount;

        for (int j = 0; j < currCount; j++)
            prev[j] = curr[j];
    }

    return resultCount;
}
