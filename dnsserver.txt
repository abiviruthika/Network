#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <sys/mman.h>
#include <signal.h>

#define MAX_DNS 100

struct DNS {
    char domain[100];
    char ip[50];
    char mac[50];
};

struct SharedDNS {
    int count;
    struct DNS table[MAX_DNS];
};

struct SharedDNS *dns;

void dns_method(int client)
{
    char domain[100] = {0};
    char ip[50] = {0};
    char mac[50] = {0};
    char response[300] = {0};
    int i, found = 0;

    if (recv(client, domain, sizeof(domain), 0) <= 0)
        return;

    domain[strcspn(domain, "\r\n")] = '\0';

    for (i = 0; i < dns->count; i++) {
        if (strcmp(dns->table[i].domain, domain) == 0) {
            found = 1;
            break;
        }
    }

    if (found) {
        snprintf(response, sizeof(response),
                 "Domain: %s\nIP: %s\nMAC: %s",
                 dns->table[i].domain,
                 dns->table[i].ip,
                 dns->table[i].mac);

        send(client, response, strlen(response) + 1, 0);
        return;
    }

    send(client, "NEW", 4, 0);

    if (recv(client, ip, sizeof(ip), 0) <= 0)
        return;

    if (recv(client, mac, sizeof(mac), 0) <= 0)
        return;

    ip[strcspn(ip, "\r\n")] = '\0';
    mac[strcspn(mac, "\r\n")] = '\0';

    if (dns->count >= MAX_DNS) {
        strcpy(response, "DNS table is full!");
    } else {
        i = dns->count;

        strcpy(dns->table[i].domain, domain);
        strcpy(dns->table[i].ip, ip);
        strcpy(dns->table[i].mac, mac);
        dns->count++;

        snprintf(response, sizeof(response),
                 "New DNS entry saved!\nDomain: %s\nIP: %s\nMAC: %s",
                 domain, ip, mac);
    }

    send(client, response, strlen(response) + 1, 0);
}

int main(int argc, char *argv[])
{
    int server, client, port, opt = 1;
    struct sockaddr_in address;
    pid_t pid;

    if (argc != 2) {
        printf("Usage: %s <port>\n", argv[0]);
        return 1;
    }

    port = atoi(argv[1]);

    dns = mmap(NULL, sizeof(struct SharedDNS),
               PROT_READ | PROT_WRITE,
               MAP_SHARED | MAP_ANONYMOUS, -1, 0);

    if (dns == MAP_FAILED) {
        perror("mmap");
        return 1;
    }

    memset(dns, 0, sizeof(struct SharedDNS));

    dns->count = 3;

    strcpy(dns->table[0].domain, "naruto.anime");
    strcpy(dns->table[0].ip, "192.168.1.10");
    strcpy(dns->table[0].mac, "AA:BB:CC:11:22:33");

    strcpy(dns->table[1].domain, "onepiece.anime");
    strcpy(dns->table[1].ip, "192.168.1.20");
    strcpy(dns->table[1].mac, "AA:BB:CC:44:55:66");

    strcpy(dns->table[2].domain, "bleach.anime");
    strcpy(dns->table[2].ip, "192.168.1.30");
    strcpy(dns->table[2].mac, "AA:BB:CC:77:88:99");

    signal(SIGCHLD, SIG_IGN);

    server = socket(AF_INET, SOCK_STREAM, 0);
    if (server < 0) {
        perror("socket");
        return 1;
    }

    setsockopt(server, SOL_SOCKET, SO_REUSEADDR,
               &opt, sizeof(opt));

    memset(&address, 0, sizeof(address));
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(port);

    if (bind(server, (struct sockaddr *)&address,
             sizeof(address)) < 0) {
        perror("bind");
        close(server);
        return 1;
    }

    if (listen(server, 5) < 0) {
        perror("listen");
        close(server);
        return 1;
    }

    printf("Anime DNS Server running on port %d\n", port);

    while (1) {
        client = accept(server, NULL, NULL);

        if (client < 0) {
            perror("accept");
            continue;
        }

        pid = fork();

        if (pid == 0) {
            close(server);
            dns_method(client);
            close(client);
            _exit(0);
        } else if (pid > 0) {
            close(client);
        } else {
            perror("fork");
            close(client);
        }
    }

    close(server);
    munmap(dns, sizeof(struct SharedDNS));
    return 0;
}
