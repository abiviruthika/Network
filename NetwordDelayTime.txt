#include <stdlib.h>
#include <limits.h>

int networkDelayTime(int** times, int timesSize,
                     int* timesColSize, int n, int k)
{
    int dist[101], visited[101] = {0};

    for (int i = 1; i <= n; i++)
        dist[i] = INT_MAX;

    dist[k] = 0;

    for (int i = 1; i <= n; i++)
    {
        int u = -1;
        int min = INT_MAX;

        for (int j = 1; j <= n; j++)
        {
            if (!visited[j] && dist[j] < min)
            {
                min = dist[j];
                u = j;
            }
        }

        if (u == -1)
            break;

        visited[u] = 1;

        for (int j = 0; j < timesSize; j++)
        {
            int from = times[j][0];
            int to = times[j][1];
            int weight = times[j][2];

            if (from == u && dist[u] + weight < dist[to])
            {
                dist[to] = dist[u] + weight;
            }
        }
    }

    int ans = 0;

    for (int i = 1; i <= n; i++)
    {
        if (dist[i] == INT_MAX)
            return -1;

        if (dist[i] > ans)
            ans = dist[i];
    }

    return ans;
}
