typedef struct
{
    int vertexList[20];
    int count;
} PQueue;

void InitializeQueue(PQueue *queue, int V)
{
    int i;
    queue->count = 0;

    for (i = 0; i < V; i++)
    {
        queue->vertexList[i] = i;
        queue->count++;
    }
}

int IsQueueEmpty(PQueue *queue)
{
    return queue->count == 0;
}

int DeleteMin(PQueue *queue, int dist[])
{
    int minPos = 0;
    int minValue = dist[queue->vertexList[0]];
    int vertex;
    int i;

    for (i = 1; i < queue->count; i++)
    {
        if (dist[queue->vertexList[i]] < minValue)
        {
            minValue = dist[queue->vertexList[i]];
            minPos = i;
        }
    }

    vertex = queue->vertexList[minPos];

    for (i = minPos; i < queue->count - 1; i++)
    {
        queue->vertexList[i] = queue->vertexList[i + 1];
    }

    queue->count--;

    return vertex;
}

void equationToBinary(char equation[],char binary[])
{
    int degree=-1;
    int powers[30];
    int count=0;
    int i,j;
    char temp[100];
    j=0;
    for(i=0;equation[i]!='\0';i++)
    {
        if(equation[i]!=' ')
            temp[j++]=equation[i];
    }
    temp[j]='\0';
    i=0;
    while(temp[i]!='\0')
    {
        if(temp[i]=='x')
        {
            if(temp[i+1]=='^')
            {
                int p=0;
                i+=2;
                while(temp[i]>='0' && temp[i]<='9')
                {
                    p=p*10+(temp[i]-'0');
                    i++;
                }
                powers[count++]=p;
                if(p>degree)
                    degree=p;
            }
            else
            {
                powers[count++]=1;
                if(degree<1)
                    degree=1;
                i++;
            }
        }
        else if(temp[i]=='1')
        {
            powers[count++]=0;
            if(degree<0)
                degree=0;
            i++;
        }
        else
        {
            i++;
        }
    }
    for(i=degree;i>=0;i--)
    {
        binary[degree-i]='0';
        for(j=0;j<count;j++)
        {
            if(powers[j]==i)
            {
                binary[degree-i]='1';
                break;
            }
        }
    }
    binary[degree+1]='\0';
}
void computeCRC(char data[],char divisor[],char remainder[])
{
    char temp[512];
    int dataLen = strlen(data);
    int divLen = strlen(divisor);
    int i, j;
    strcpy(temp, data);
    for(i = 0; i < divLen - 1; i++)
        temp[dataLen + i] = '0';
    temp[dataLen + divLen - 1] = '\0';
    for(i = 0; i < dataLen; i++)
    {
        if(temp[i] == '1')
        {
            for(j = 0; j < divLen; j++)
            {
                if(temp[i + j] == divisor[j])
                    temp[i + j] = '0';
                else
                    temp[i + j] = '1';
            }
        }
    }
    for(i = 0; i < divLen - 1; i++)
        remainder[i] = temp[dataLen + i];
    remainder[divLen - 1] = '\0';
}

void writeFrameStreamToFile(const char *filename, int totalPackets, int charSize, int totalLen, int frameBitSize,
                            const char *buffer, char divisor[], const char *srcMacBin, const char *destMacBin,
                            const char *ipBinSrc, const char *ipBinDest, const char *srcPortBin, const char *destPortBin)
{
    FILE *outFile = fopen("output.txt", "w");
    if (outFile == NULL)
    {
        printf("Error: Could not create output file %s!\n", filename);
        return;
    }
    fprintf(outFile, "----------------------------------------\n");
    fprintf(outFile, " COMPLETE BINARY FRAME STREAM OUTPUT\n");
    fprintf(outFile, "------------------------------------------\n\n");
    int currentFrameNum = 1;
    int pIdx, frameInPacket, b;
    for (pIdx = 0; pIdx < totalPackets; pIdx++) {
        int byteStart = pIdx * charSize;
        int bytesInPacket = charSize;
        if (byteStart + bytesInPacket > totalLen)
        {
            bytesInPacket = totalLen - byteStart;
        }
        int bitsInPacket = bytesInPacket * 8;
        int framesForPacket = (bitsInPacket + (frameBitSize - 1)) / frameBitSize;
        for (frameInPacket = 1; frameInPacket <= framesForPacket; frameInPacket++)
        {
            int payloadBits = frameBitSize;
            int paddingBits = 0;
            char payloadString[128];
            int bitStart = (frameInPacket - 1) * frameBitSize;
            if (bitStart + payloadBits > bitsInPacket)
            {
                payloadBits = bitsInPacket - bitStart;
                paddingBits = frameBitSize - payloadBits;
            }
            for (b = 0; b < payloadBits; b++) {
                int BitPos = bitStart + b;
                int charset = BitPos / 8;
                int bitShift = 7 - (BitPos % 8);
                int bitVal = (buffer[byteStart + charset] >> bitShift) & 1;
                payloadString[b] = bitVal ? '1' : '0';
            }
            payloadString[payloadBits] = '\0';
            char fullPayloadWithPadding[256];
            strcpy(fullPayloadWithPadding, payloadString);
            for(b = 0; b < paddingBits; b++) {
                strcat(fullPayloadWithPadding, "0");
            }
            char crcRemainder[32];
            computeCRC(fullPayloadWithPadding, divisor, crcRemainder);
            fprintf(outFile, "Frame %d Stream: %s %s %s %s %s %s %s",
                    currentFrameNum, srcMacBin, destMacBin, ipBinSrc, ipBinDest, srcPortBin, destPortBin, payloadString);
            for(b = 0; b < paddingBits; b++) {
                fprintf(outFile, "0");
            }
            fprintf(outFile, " %s\n", crcRemainder);
            currentFrameNum++;
        }
    }
    fclose(outFile);
}
int Binary(int num, char *str)
{
    int temp = num;
    int i;
    for(i = 0; i < 8; i++)
    {
        str[i] = '0';
    }
    str[8] = '\0';
    for (i = 7; i >= 0; i--)
    {
        if (temp % 2 == 1)
        {
            str[i] = '1';
        } else {
            str[i] = '0';
        }
        temp = temp / 2;
    }
    return 1;
}

int macToBinary(const char *mac, char *str)
{
    int len = (int)strlen(mac);
    int Idx = 0;
    int Count = 0;
    int i, j;
    for (i = 0; i < len; i++)
    {
        char c;
        int digit = 0;
        int temp;
        if (mac[i] == ':')
        {
            continue;
        }
        c = mac[i];
        if (c >= '0' && c <= '9') digit = c - '0';
        else if (c >= 'A' && c <= 'F') digit = c - 'A' + 10;
        else if (c >= 'a' && c <= 'f') digit = c - 'a' + 10;
        temp = digit;
        for (j = 3; j >= 0; j--)
        {
            if (temp % 2 == 1)
            {
                str[Idx + j] = '1';
            }
            else
            {
                str[Idx + j] = '0';
            }
            temp = temp / 2;
        }
        Idx += 4;
        Count++;
        if (Count % 2 == 0 && i < len - 1) {
            str[Idx] = ' ';
            Idx++;
        }
    }
    str[Idx] = '\0';
    return 1;
}

static void trim(char *s) {
    int i = 0, j = (int)strlen(s) - 1;
    while (i <= j && isspace((unsigned char)s[i])) i++;
    while (j >= i && isspace((unsigned char)s[j])) j--;
    if (i == 0 && j == (int)strlen(s) - 1) return;
    int dst = 0;
    int k;
    for (k = i; k <= j; k++) s[dst++] = s[k];
    s[dst] = '\0';

static void remove_spaces_inplace(char *s) {
    int r = 0, w = 0;
    while (s[r]) {

}
void computeCRC(const char *inputBits, const char *generator, char *remainder) {
    int genLen = (int)strlen(generator);
    int inputLen = (int)strlen(inputBits);
    if (genLen < 2 || inputLen < genLen - 1) {
        remainder[0] = '\0';
        return;
    }
    if (inputLen >= MAX_BITS - 1) {
        remainder[0] = '\0';
        return;
    }
    strcpy(workData, inputBits);
    int i, j;
    for (i = 0; i <= inputLen - genLen; i++) {
        if (workData[i] == '1') {
            for (j = 0; j < genLen; j++) {
                workData[i + j] = (workData[i + j] == generator[j]) ? '0' : '1';
            }
        }
    }
    if (genLen - 1 > 0) {
        strncpy(remainder, workData + (inputLen - genLen + 1), genLen - 1);
        remainder[genLen - 1] = '\0';
    } else {
        remainder[0] = '\0';
    }
}

    while (*p && (*p == '0' || *p == '1') && idx < outSize - 1) {
        outGen[idx++] = *p++;
    }
    outGen[idx] = '\0';

}
static int parse_frame_stream_line(const char *ln, char *outFullStream, int fullSize,
                                   int *outPayloadStart, int *outPayloadLen,
                                   int *outCrcStart, int *outCrcLen) {
    char tmp[MAX_BITS];
    char *tokens[256];
    int tcount = 0;
    char *tok;
    strncpy(tmp, ln, sizeof(tmp) - 1);
    tmp[sizeof(tmp) - 1] = '\0';
    tok = strtok(tmp, " \t\r\n");
    while (tok != NULL && tcount < (int)(sizeof(tokens)/sizeof(tokens[0]))) {
        tokens[tcount++] = tok;
        tok = strtok(NULL, " \t\r\n");
    }
    if (tcount >= 5) {
        int startIdx = 3; /* token index where bit tokens start */
        int i;
        outFullStream[0] = '\0';
        for (i = startIdx; i < tcount; i++) {
            char tokenCopy[MAX_STR];
            strncpy(tokenCopy, tokens[i], sizeof(tokenCopy) - 1);
            tokenCopy[sizeof(tokenCopy) - 1] = '\0';
            remove_spaces_inplace(tokenCopy);
            strncat(outFullStream, tokenCopy, fullSize - strlen(outFullStream) - 1);
        }
        char payloadCopy[MAX_BITS], crcCopy[MAX_STR];
        strncpy(payloadCopy, tokens[tcount - 2], sizeof(payloadCopy) - 1);
        payloadCopy[sizeof(payloadCopy) - 1] = '\0';
        strncpy(crcCopy, tokens[tcount - 1], sizeof(crcCopy) - 1);
        crcCopy[sizeof(crcCopy) - 1] = '\0';
        remove_spaces_inplace(payloadCopy);
        remove_spaces_inplace(crcCopy);
        int pos = 0;
        for (i = startIdx; i < tcount - 2; i++) {
            char tokenCopy[MAX_STR];
            strncpy(tokenCopy, tokens[i], sizeof(tokenCopy)-1);
            tokenCopy[sizeof(tokenCopy)-1] = '\0';
            remove_spaces_inplace(tokenCopy);
            pos += (int)strlen(tokenCopy);
        }
        *outPayloadStart = pos;
        *outPayloadLen = (int)strlen(payloadCopy);
        *outCrcStart = pos + *outPayloadLen;
        *outCrcLen = (int)strlen(crcCopy);
        return 1;
    }
    return 0;
}

{
    char high[9],low[9];

        str[index++]=high[i];

{
    int len=strlen(mac);
    int index=0,count=0;
    int i,j;
    for(i=0;i<len;i++)
    {
        int value,temp;
        if(mac[i]==':')

        if(mac[i]>='0' && mac[i]<='9')
            value=mac[i]-'0';
        else if(mac[i]>='A' && mac[i]<='F')
            value=mac[i]-'A'+10;
        else
            value=mac[i]-'a'+10;
        temp=value;
        for(j=3;j>=0;j--)
        {
            if(temp%2)
                str[index+j]='1';
            else
                str[index+j]='0';
            temp/=2;
        }
        index+=4;
        count++;
        if(count%2==0 && i<len-1)
            str[index++]=' ';
    }
    str[index]='\0';
    return 1;
}

unsigned int binary(const char *binary_str, int start, int length) {
    unsigned int val = 0;
    int i;
    for (i = 0; i < length; i++) {
        val <<= 1;
        if (binary_str[start + i] == '1') {
            val |= 1;
        }
    }
    return val;
}
void print_as_binary(unsigned int value, int bit_width) {
    int i;
    for (i = bit_width - 1; i >= 0; i--) {
        printf("%d", (value >> i) & 1);
    }
}

}

{

    {
        char c;
        int digit = 0;
        int temp;
        if (mac[i] == ':')
        {
            continue;
        }
        c = mac[i];
        if (c >= '0' && c <= '9') digit = c - '0';
        else if (c >= 'A' && c <= 'F') digit = c - 'A' + 10;
        else if (c >= 'a' && c <= 'f') digit = c - 'a' + 10;
        temp = digit;
        for (j = 3; j >= 0; j--)

            {
                str[Idx + j] = '0';

        }

int printDataLinkLabel(int sIdx, int dIdx, const char *buffer, int totalLen, int srcPort, int packetBitSize, int frameBitSize) {
    int i;
    int charSize = packetBitSize / 8;
    if (charSize < 1) charSize = 2;
    int totalPackets = (totalLen + (charSize - 1)) / charSize;
    int grandTotalFrames = 0;

    char ipBinSrc[50], ipBinDest[50], temp[9];

    int currentFrameNum = 1;
    int pIdx, frameInPacket;
    printf("\n-------------------------------\n");
    printf(" LAYER 4 : DATA LINK LAYER \n");
    printf("---------------------------------\n");
    for (pIdx = 0; pIdx < totalPackets; pIdx++) {
        int byteStart = pIdx * charSize;
        int bytesInPacket = charSize;
        if (byteStart + bytesInPacket > totalLen) {
            bytesInPacket = totalLen - byteStart;
        }
        int bitsInPacket = bytesInPacket * 8;
        int framesForPacket = (bitsInPacket + (frameBitSize - 1)) / frameBitSize;
        printf("\n--------------------------------------------------------------\n");
        printf("PACKET %d (Contains %d Bits)\n", pIdx + 1, bitsInPacket);
        printf("--------------------------------------------------------------\n");
        for (frameInPacket = 1; frameInPacket <= framesForPacket; frameInPacket++) {
            int payloadBits = frameBitSize;
            int paddingBits = 0;
            char payloadString[128];
            int bitStart = (frameInPacket - 1) * frameBitSize;
            if (bitStart + payloadBits > bitsInPacket) {
                payloadBits = bitsInPacket - bitStart;
                paddingBits = frameBitSize - payloadBits;
            }
            int b;
            for (b = 0; b < payloadBits; b++) {
                int BitPos = bitStart + b;
                int charset = BitPos / 8;
                int bitShift = 7 - (BitPos % 8);
                int bitVal = (buffer[byteStart + charset] >> bitShift) & 1;
                payloadString[b] = bitVal ? '1' : '0';
            }
            payloadString[payloadBits] = '\0';

            printf("  Payload Bits    : %d\n", payloadBits);
            printf("  Payload Text    : %s\n", payloadString);
            printf("  Padding Bits    : %d\n", paddingBits);
            if (paddingBits > 0) {
                printf("  Padding Content : ");
                for(b = 0; b < paddingBits; b++) printf("0");
                printf("\n");
            }
            printf("  Error Control   : 00000000\n");
            currentFrameNum++;
            grandTotalFrames++;
        }
    }
    printf("\n----------------------------------------\n");
    printf(" COMPLETE BINARY FRAME STREAM OUTPUT\n");
    printf("------------------------------------------\n\n");
    currentFrameNum = 1;
    for (pIdx = 0; pIdx < totalPackets; pIdx++) {
        int byteStart = pIdx * charSize;
        int bytesInPacket = charSize;
        if (byteStart + bytesInPacket > totalLen) {
            bytesInPacket = totalLen - byteStart;
        }
        int bitsInPacket = bytesInPacket * 8;
        int framesForPacket = (bitsInPacket + (frameBitSize - 1)) / frameBitSize;
        for (frameInPacket = 1; frameInPacket <= framesForPacket; frameInPacket++) {
            int payloadBits = frameBitSize;

            char payloadString[128];
            int bitStart = (frameInPacket - 1) * frameBitSize;
            if (bitStart + payloadBits > bitsInPacket) {
                payloadBits = bitsInPacket - bitStart;
                paddingBits = frameBitSize - payloadBits;
            }
            int b;
            for (b = 0; b < payloadBits; b++) {
                int BitPos = bitStart + b;
                int charset = BitPos / 8;
                int bitShift = 7 - (BitPos % 8);
                int bitVal = (buffer[byteStart + charset] >> bitShift) & 1;
                payloadString[b] = bitVal ? '1' : '0';
            }
            payloadString[payloadBits] = '\0';
            printf("Frame %d Stream: %s %s %s %s %s %s %s", currentFrameNum, srcMacBin, destMacBin, ipBinSrc, ipBinDest, srcPortBin, destPortBin, payloadString);

            printf(" 00000000\n");

    };

}
