#include <stdlib.h>
#include <stdbool.h>

int minReorder(int n, int** connections, int connectionsSize, int* connectionsColSize)
{
    int *head = (int*)malloc(n * sizeof(int));
    int *to = (int*)malloc(2 * (n - 1) * sizeof(int));
    int *next = (int*)malloc(2 * (n - 1) * sizeof(int));
    int *cost = (int*)malloc(2 * (n - 1) * sizeof(int));

    for (int i = 0; i < n; i++)
        head[i] = -1;

    int idx = 0;

    for (int i = 0; i < connectionsSize; i++)
    {
        int u = connections[i][0];
        int v = connections[i][1];

        to[idx] = v;
        cost[idx] = 1;
        next[idx] = head[u];
        head[u] = idx++;

        to[idx] = u;
        cost[idx] = 0;
        next[idx] = head[v];
        head[v] = idx++;
    }

    bool *visited = (bool*)calloc(n, sizeof(bool));

    int *queue = (int*)malloc(n * sizeof(int));
    int front = 0, rear = 0;
    int count = 0;

    queue[rear++] = 0;
    visited[0] = true;

    while (front < rear)
    {
        int u = queue[front++];

        for (int i = head[u]; i != -1; i = next[i])
        {
            int v = to[i];

            if (!visited[v])
            {
                visited[v] = true;
                count += cost[i];
                queue[rear++] = v;
            }
        }
    }

    free(head);
    free(to);
    free(next);
    free(cost);
    free(visited);
    free(queue);

    return count;
}
