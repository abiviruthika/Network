#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main(int argc, char *argv[])
{
    int sockfd, connfd;
    struct sockaddr_in server, client;
    socklen_t len;
    char ip[50];
    char mac[50];

    if (argc != 2) {
        printf("Usage: %s <port>\n", argv[0]);
        return 1;
    }

    sockfd = socket(AF_INET, SOCK_STREAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(atoi(argv[1]));
    server.sin_addr.s_addr = INADDR_ANY;

    bind(sockfd, (struct sockaddr *)&server, sizeof(server));
    listen(sockfd, 5);

    printf("ARP Server waiting...\n");

    len = sizeof(client);
    connfd = accept(sockfd, (struct sockaddr *)&client, &len);

    recv(connfd, ip, sizeof(ip) - 1, 0);
    ip[sizeof(ip) - 1] = '\0';

    printf("Received IP Address: %s\n", ip);

    if (strcmp(ip, "192.168.1.10") == 0)
        strcpy(mac, "AA:BB:CC:DD:EE:10");
    else if (strcmp(ip, "192.168.1.20") == 0)
        strcpy(mac, "AA:BB:CC:DD:EE:20");
    else
        strcpy(mac, "MAC Address Not Found");

    send(connfd, mac, strlen(mac) + 1, 0);

    printf("MAC Address sent: %s\n", mac);

    close(connfd);
    close(sockfd);

    return 0;
}
