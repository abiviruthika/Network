#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

int main(int argc, char *argv[])
{
    int sock, port;
    struct sockaddr_in server;
    char domain[100], ip[50], mac[50], response[300];
    ssize_t n;

    if (argc != 3) {
        printf("Usage: %s <server_ip> <port>\n", argv[0]);
        return 1;
    }

    port = atoi(argv[2]);

    sock = socket(AF_INET, SOCK_STREAM, 0);
    if (sock < 0) {
        perror("socket");
        return 1;
    }

    memset(&server, 0, sizeof(server));
    server.sin_family = AF_INET;
    server.sin_port = htons(port);

    if (inet_pton(AF_INET, argv[1], &server.sin_addr) != 1) {
        printf("Invalid server IP\n");
        close(sock);
        return 1;
    }

    if (connect(sock, (struct sockaddr *)&server,
                sizeof(server)) < 0) {
        perror("connect");
        close(sock);
        return 1;
    }

    printf("===== ANIME DNS CLIENT =====\n");
    printf("Enter anime domain: ");

    if (scanf("%99s", domain) != 1) {
        close(sock);
        return 1;
    }

    if (send(sock, domain, strlen(domain) + 1, 0) < 0) {
        perror("send");
        close(sock);
        return 1;
    }

    memset(response, 0, sizeof(response));
    n = recv(sock, response, sizeof(response) - 1, 0);

    if (n <= 0) {
        printf("No response from server\n");
        close(sock);
        return 1;
    }

    if (strcmp(response, "NEW") == 0) {
        printf("New domain! Enter its details.\n");

        printf("Enter IP address: ");
        scanf("%49s", ip);

        printf("Enter MAC address: ");
        scanf("%49s", mac);

        send(sock, ip, strlen(ip) + 1, 0);
        send(sock, mac, strlen(mac) + 1, 0);

        memset(response, 0, sizeof(response));
        n = recv(sock, response, sizeof(response) - 1, 0);

        if (n > 0)
            printf("\n%s\n", response);
    } else {
        printf("\nDNS Response:\n%s\n", response);
    }

    close(sock);
    return 0;
}
